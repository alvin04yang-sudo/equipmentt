function doGet(e) { var equipmentId = e.parameter.id; var action = e.parameter.action || "borrow"; if (!equipmentId) { return ContentService.createTextOutput("請提供設備 ID，例如：?id=1001&action=borrow"); } var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Equipment_List"); var data = sheet.getDataRange().getValues(); var responseMessage = ""; var isUpdated = false; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(equipmentId).trim()) { var currentStatus = String(data[i][2]).trim(); var eqName = data[i][1]; var eqBrand = data[i][4]; // 如果動作是借用 if (action === "borrow") { if (currentStatus === "可借用") { sheet.getRange(i + 1, 3).setValue("借出中"); responseMessage = "借用成功！\n" + eqBrand + " " + eqName + "\n狀態已更新為：借出中"; isUpdated = true; } else { return ContentService.createTextOutput("借用失敗！該設備目前狀態為：" + currentStatus); } } // 如果動作是歸還 else if (action === "return") { if (currentStatus === "借出中") { sheet.getRange(i + 1, 3).setValue("可借用"); responseMessage = "歸還成功！\n" + eqBrand + " " + eqName + "\n狀態已更新為：可借用"; isUpdated = true; } else { return ContentService.createTextOutput("歸還失敗！該設備目前狀態為：" + currentStatus + "（未在借出狀態）"); } } break; // 找到對應設備後即可跳出迴圈 } } if (!responseMessage) { return ContentService.createTextOutput("找不到設備編號：" + equipmentId); } // 若成功更新狀態，自動更新「已借出清單」分頁 if (isUpdated) { updateBorrowedListSheet(ss); } return ContentService.createTextOutput(responseMessage); } // 📌 自動將「借出中」的設備更新至 Borrowed_List 分頁 function updateBorrowedListSheet(ss) { var sourceSheet = ss.getSheetByName("Equipment_List"); var targetSheet = ss.getSheetByName("Borrowed_List"); // 若分頁不存在則自動建立 if (!targetSheet) { targetSheet = ss.insertSheet("Borrowed_List"); } // 清空目標分頁舊資料 targetSheet.clear(); var data = sourceSheet.getDataRange().getValues(); var borrowedData = []; // 複製標題列 (第一列) if (data.length > 0) { borrowedData.push(data[0]); } // 篩選出狀態為「借出中」的資料 for (var i = 1; i < data.length; i++) { var status = String(data[i][2]).trim(); if (status === "借出中") { borrowedData.push(data[i]); } } // 寫入目標分頁 if (borrowedData.length > 0) { targetSheet.getRange(1, 1, borrowedData.length, borrowedData[0].length).setValues(borrowedData); } }
